cmake_minimum_required(VERSION 3.16)
project(mini_lla LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(mini_lla_core src/tensor.cc src/ops.cc src/matmul_optimized.cc)
target_include_directories(mini_lla_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
  target_compile_options(mini_lla_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(mini_lla_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(mini-lla-tests tests/test_main.cc tests/test_tensor.cc tests/test_ops.cc)
target_link_libraries(mini-lla-tests PRIVATE mini_lla_core)
if(MSVC)
  target_compile_options(mini-lla-tests PRIVATE /W4 /utf-8)
else()
  target_compile_options(mini-lla-tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME mini-lla-tests COMMAND mini-lla-tests)
